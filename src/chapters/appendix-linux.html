Lab 2: Bash Shell Scripting (`check_number.sh`)

#!/bin/bash
# lab2_script.sh - Kiểm tra số nguyên tố và tính tổng
read -p "Nhập số nguyên N: " N

if [ $N -le 0 ]; then
    echo "Vui lòng nhập số nguyên dương!"
    exit 1
fi

sum=0
for (( i=1; i<=N; i++ )); do
    sum=$((sum + i))
done

echo "Tổng từ 1 đến $N là: $sum"

Lab 3: Tiến Trình & Bộ Nhớ Chia Sẻ (Shared Memory)

/* lab3_shm.c - Sử dụng POSIX Shared Memory shm_open & mmap có kiểm tra lỗi */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    const int SIZE = 4096;
    const char *name = "IT007_Shm";

    int fd = shm_open(name, O_CREAT | O_RDWR, 0666);
    if (fd < 0) { perror("shm_open failed"); exit(1); }
    ftruncate(fd, SIZE);

    void *ptr = mmap(0, SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (ptr == MAP_FAILED) { perror("mmap failed"); exit(1); }

    pid_t pid = fork();
    if (pid < 0) { perror("fork failed"); exit(1); }
    
    if (pid == 0) { // Child Writes
        sprintf((char*)ptr, "Hello from Child Process! PID = %d", getpid());
        exit(0);
    } else { // Parent Reads
        wait(NULL);
        printf("Parent read from Shared Memory: %s\n", (char*)ptr);
        munmap(ptr, SIZE);
        shm_unlink(name);
    }
    return 0;
}

Lab 4: Mô Phỏng Định Thời CPU (FCFS Scheduler Struct)

/* lab4_fcfs.c - Struct dữ liệu tính WT, TAT định thời */
typedef struct {
    int id, at, bt, ct, tat, wt;
} Process;

void calculate_fcfs(Process p[], int n) {
    int current_time = 0;
    for (int i = 0; i < n; i++) {
        if (current_time < p[i].at) current_time = p[i].at;
        p[i].ct = current_time + p[i].bt;
        p[i].tat = p[i].ct - p[i].at;
        p[i].wt = p[i].tat - p[i].bt;
        current_time = p[i].ct;
    }
}

Lab 5: Đồng Bộ POSIX Threads Với Semaphore (`pthread_sem.c`)

/* lab5_pthread.c - Độc quyền truy xuất Mutex bằng POSIX Semaphore */
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

sem_t mutex;
int counter = 0;

void* worker(void* arg) {
    sem_wait(&mutex);
    counter++;
    printf("Thread %ld incremented counter to %d\n", (long)arg, counter);
    sem_post(&mutex);
    return NULL;
}

int main() {
    pthread_t threads[5];
    sem_init(&mutex, 0, 1);
    for (long i = 0; i < 5; i++) pthread_create(&threads[i], NULL, worker, (void*)i);
    for (int i = 0; i < 5; i++) pthread_join(threads[i], NULL);
    sem_destroy(&mutex);
    return 0;
}

Lab 6: Simple Shell Tích Hợp Pipe (`lab6_pipe.c` - Complete Program)

/* lab6_pipe.c - Chương trình biên dịch hoàn chỉnh thực thi lệnh: ls -l | grep txt */
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    char *cmd1[] = {"ls", "-l", NULL};
    char *cmd2[] = {"grep", "txt", NULL};

    int pipefd[2];
    if (pipe(pipefd) < 0) { perror("pipe failed"); exit(1); }

    pid_t p1 = fork();
    if (p1 < 0) { perror("fork 1 failed"); exit(1); }

    if (p1 == 0) { // Child 1: Chạy 'ls -l' và ghi vào pipe
        close(pipefd[0]); // Đóng đầu đọc không dùng
        if (dup2(pipefd[1], STDOUT_FILENO) < 0) { perror("dup2 failed"); exit(1); }
        close(pipefd[1]);
        execvp(cmd1[0], cmd1);
        perror("execvp cmd1 failed");
        exit(1);
    }

    pid_t p2 = fork();
    if (p2 < 0) { perror("fork 2 failed"); exit(1); }

    if (p2 == 0) { // Child 2: Chạy 'grep txt' và đọc từ pipe
        close(pipefd[1]); // Đóng đầu ghi không dùng
        if (dup2(pipefd[0], STDIN_FILENO) < 0) { perror("dup2 failed"); exit(1); }
        close(pipefd[0]);
        execvp(cmd2[0], cmd2);
        perror("execvp cmd2 failed");
        exit(1);
    }

    // Tiến trình cha đóng cả 2 đầu pipe và chờ 2 con hoàn thành
    close(pipefd[0]);
    close(pipefd[1]);
    waitpid(p1, NULL, 0);
    waitpid(p2, NULL, 0);
    return 0;
}

  • Luôn kiểm tra mã lỗi trả về của tất cả System Calls (`if (pid < 0)`, `if (fd < 0)`).
  • Dọn dẹp tài nguyên (dùng `wait()`, `munmap()`, `shm_unlink()`, `close()`) để tránh rò rỉ RAM và Zombie process.
